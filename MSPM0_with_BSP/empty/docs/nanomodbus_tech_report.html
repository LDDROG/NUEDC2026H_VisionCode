2. 分层架构

职责划分

层文件职责
应用层empty.c业务逻辑、主循环轮询
协议层nanomodbus.cModbus RTU 协议解析/构造
移植层port.c硬件桥接、数据模型
队列层nmbs_ring_queue.cISR↔协议层解耦
BSP 层bsp_usart.cUART/DMA 硬件抽象
驱动层DL_xxx寄存器级操作

设计原则

  • 传输无关：核心库通过 read/write 函数指针与硬件解耦
  • ISR 最小化：ISR 只做 memcpy 到队列，协议解析在主循环
  • SPSC 无锁：环形队列单生产者(ISR)单消费者(主循环)，无需互斥
  • 零拷贝发送：DMA 直接从发送缓冲区搬运，无需中间拷贝

7. 客户端（主机）架构总览

单片机作为 Modbus RTU 主机（Client/Master）时，主动发起请求并等待从机响应。与服务端的核心区别：

服务端（从机）

  • 被动等待请求
  • 回调模式：收到请求 → 执行回调 → 自动响应
  • 数据模型：nmbs_server_t（线圈+寄存器）
  • 主循环调用 nmbs_server_poll()

客户端（主机）

  • 主动发起请求
  • 命令模式：调用 API → 发送请求 → 等待响应
  • 无数据模型，直接操作 API 返回值
  • 应用代码按需调用读写函数

10. 客户端发送路径

write_serial 的客户端特殊行为

当 ctx->is_client == 1 时，write_serial() 在发送前会执行一个关键操作：

if (ctx->is_client)
{
    NMBSRingQueueReset(&ctx->rx_queue);  /* 清除残留数据 */
}

原因：Modbus RTU 是半双工协议，主机发送请求前，总线上可能残留之前的响应帧或噪声数据。清空队列确保 read_serial() 读到的都是本次请求的响应。

15. 使用示例

示例 1：服务端（从机）— 最简配置

#include "ti_msp_dl_config.h"
#include "bsp_log.h"
#include "bsp_time.h"
#include "nanoMODBUS/nmbs/port.h"

static nmbs_t g_nmbs;
static nmbs_server_t g_server = {
    .id = 0x01,             /* RTU 从机地址 */
    .coils = {0},
    .regs  = {0},
};

int main(void)
{
    SYSCFG_DL_init();
    NVIC_EnableIRQ(UART_0_INST_INT_IRQN);
    NVIC_EnableIRQ(TIMER_0_INST_INT_IRQN);
    TimeInit();
    BSPLogInit();

    /* 预设寄存器值 */
    g_server.regs[0] = 0x1234;   /* 设备 ID */
    g_server.regs[1] = 0;        /* 运行秒数 */

    /* 初始化服务端 (config=NULL → 默认 UART0 + DMA) */
    nmbs_error err = nmbs_server_init(&g_nmbs, &g_server, NULL);
    if (err != NMBS_ERROR_NONE) {
        LOGERROR("init failed: %d", (int)err);
        while (1);
    }

    LOGINFO("Modbus slave id=%d started", g_server.id);

    uint32_t last_ms = TimeGetBootMs();

    while (1) {
        /* 轮询 Modbus 请求 */
        nmbs_server_poll(&g_nmbs);

        /* 每秒更新心跳寄存器 */
        if (TimeGetBootMs() - last_ms >= 1000) {
            last_ms = TimeGetBootMs();
            g_server.regs[1]++;
            DL_GPIO_togglePins(LED_PORT, LED_PIN_0_PIN);
        }
    }
}

/* 1ms 定时中断驱动帧间隙判帧 */
void TIMER_0_INST_IRQHandler(void)
{
    DL_TimerG_clearInterruptStatus(TIMER_0_INST, DL_TIMERG_INTERRUPT_ZERO_EVENT);
    USARTServiceTickUs(1000U);
}

示例 2：服务端 — 自定义配置

/* 使用 UART1 + 自定义 DMA 通道 + 自定义超时 */
nmbs_port_config_t cfg = {
    .usart           = NULL,       /* 自动注册 */
    .uart_handle_id  = 1,          /* UART1 */
    .dma_rx_channel  = DMA_CH3_CHAN_ID,
    .dma_tx_channel  = DMA_CH2_CHAN_ID,
    .rx_frame_gap_us = 3000,       /* 3ms 帧间隙 */
    .byte_timeout_ms = 200,        /* 200ms 字节超时 */
    .read_timeout_ms = 2000,       /* 2s 读超时 */
};

nmbs_error err = nmbs_server_init(&g_nmbs, &g_server, &cfg);

示例 3：服务端 — 绑定外部已注册的 USART

/* 先注册 USART (例如与其他模块共享 UART) */
USART_Init_Config_s usart_cfg = {
    .recv_buff      = my_buffer,
    .recv_buff_size = 256,
    .usart_handle   = UART_0_INST,
    .module_callback = my_other_callback,  /* 会被链式保留 */
    .dma_rx_channel = DMA_CH1_CHAN_ID,
    .dma_tx_channel = DMA_CH0_CHAN_ID,
    .rx_frame_gap_us = 2000,
};
USARTInstance *my_usart = USARTRegister(&usart_cfg);

/* 将已注册的 USART 传给 Modbus port */
nmbs_port_config_t cfg = {
    .usart = my_usart,  /* 外部实例 */
};

nmbs_error err = nmbs_server_init(&g_nmbs, &g_server, &cfg);
/* my_other_callback 会在 Modbus 收帧后被链式调用 */

示例 4：客户端（主机）

static nmbs_t g_client;

/* 初始化客户端 */
nmbs_error err = nmbs_client_init(&g_client, NULL);
if (err != NMBS_ERROR_NONE) { /* 处理错误 */ }

/* 设置目标从机地址 */
nmbs_set_destination_rtu_address(&g_client, 0x01);

/* 读取从机保持寄存器 [0..3] */
uint16_t regs[4] = {0};
err = nmbs_read_holding_registers(&g_client, 0, 4, regs);
if (err == NMBS_ERROR_NONE) {
    LOGINFO("regs: %04X %04X %04X %04X", regs[0], regs[1], regs[2], regs[3]);
}

/* 写单个寄存器 */
err = nmbs_write_single_register(&g_client, 10, 0xABCD);

/* 读线圈 */
nmbs_bitfield coils = {0};
err = nmbs_read_coils(&g_client, 0, 16, coils);

示例 5：应用层读写寄存器

/* 服务端回调中，应用层可直接读写 g_server.regs[] / g_server.coils[] */

/* 写入传感器数据供主机读取 */
g_server.regs[100] = temperature;   /* 温度值 */
g_server.regs[101] = humidity;      /* 湿度值 */
g_server.regs[102] = pressure;      /* 气压值 */

/* 读取主机写入的控制命令 */
uint16_t cmd = g_server.regs[200];
if (cmd == 0x0001) {
    /* 执行动作 A */
} else if (cmd == 0x0002) {
    /* 执行动作 B */
}

/* 线圈操作 (位级) */
nmbs_bitfield_set(g_server.coils, 0);     /* 置位线圈 0 */
nmbs_bitfield_unset(g_server.coils, 1);   /* 清除线圈 1 */
bool val = nmbs_bitfield_read(g_server.coils, 2);  /* 读取线圈 2 */

寄存器地址映射建议

地址范围用途示例
regs[0-9]设备信息(只读)固件版本、设备 ID、硬件版本
regs[10-99]系统状态(只读)运行时间、错误计数、温度
regs[100-199]传感器数据(只读)温度、湿度、气压、IMU
regs[200-299]控制命令(读写)启停、模式、目标值
regs[300-2047]扩展/预留—
coils[0-63]状态标志(只读)报警、就绪、运行中
coils[64-127]控制开关(读写)使能、复位、触发
